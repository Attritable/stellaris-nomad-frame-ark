namespace = nomad_frame_ark

# Potato start: mark empire + capital with ark flags (no Giga Frame planet swap)
country_event = {
	id = nomad_frame_ark.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_origin = origin_nomad_frame_ark
	}

	immediate = {
		set_country_flag = nomad_frame_ark_empire
		set_country_flag = has_ark_capital

		# Confirmed keys (do NOT misuse):
		# - giga_has_frameworld_origin is a TRIGGER (has_origin = origin_frameworld), not a flag
		# - pc_giga_frameworld is Frame World's stationary capital class — do not change_pc here
		# - Nomadic capital is arkship (civilian/science/military_arkship_tier_*) per
		#   09_scripted_triggers_nomads.txt / 29_nomad_arkships.txt
		# TODO(product): spawn/flag capital arkship megastructure if start lacks one —
		#   checked nomads_effects.txt create_arkship_effect / embark patterns; not wired yet

		if = {
			limit = { exists = capital_scope }
			capital_scope = {
				set_planet_flag = nomad_frame_ark_capital
				set_planet_flag = ark_capital
			}
		}

		if = {
			limit = { is_ai = no }
			country_event = { id = nomad_frame_ark.2 days = 1 }
		}
	}
}
